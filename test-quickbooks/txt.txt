// lib/report-template.ts

export function generateReportHTML(assets: any[], performances: any[], reportDate: string): string {
  // Group performance data by product
  const perfByProduct: Record<string, any[]> = {};
  for (const p of performances) {
    if (!perfByProduct[p.product_type]) perfByProduct[p.product_type] = [];
    perfByProduct[p.product_type].push(p);
  }

  // Portfolio totals
  const totalBeginning = assets.reduce((sum, a) => sum + (a.beginning_value || 0), 0);
  const totalEnding = assets.reduce((sum, a) => sum + (a.ending_value || 0), 0);
  const totalGainLoss = totalEnding - totalBeginning;
  const totalGainLossYTD = assets.reduce((sum, a) => sum + (a.gain_loss_ytd || 0), 0);

  const colors = ['#14B8A6', '#3B82F6', '#F59E0B', '#8B5CF6', '#EC4899', '#06B6D4', '#10B981', '#F97316'];

  const formatCurrency = (value: number) =>
    value == null ? '—' : new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', minimumFractionDigits: 0 }).format(value);
  const formatSigned = (value: number) =>
    value == null ? '—' : new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD', signDisplay: 'always' }).format(value);
  const formatNumber = (value: number) => value?.toLocaleString() ?? '—';

  return `
    <!DOCTYPE html>
    <html>
    <head>
      <meta charset="UTF-8">
      <title>Sankore Investment Report – ${reportDate}</title>
      <script src="https://cdn.jsdelivr.net/npm/chart.js@4.4.0/dist/chart.umd.min.js"></script>
      <style>
        * {
          margin: 0;
          padding: 0;
          box-sizing: border-box;
        }

        body {
          font-family: 'Inter', 'Segoe UI', Arial, sans-serif;
          background: #0A0C10;
          color: #EAEAEF;
          line-height: 1.5;
          padding: 32px 40px;
        }

        /* Print optimisation */
        @media print {
          body {
            background: white;
            color: black;
            padding: 0;
            margin: 0.75in;
          }
          .no-print-bg {
            background: none !important;
          }
          .card, .product-tag, th, .date-badge {
            background: #f0f0f0 !important;
            color: black !important;
            border-color: #ccc !important;
          }
          .card-value, .positive, .negative, h2, h3, h1 {
            color: black !important;
            background: none !important;
            -webkit-text-fill-color: black !important;
            text-shadow: none !important;
          }
          .product-page {
            page-break-before: always;
          }
          .chart-container canvas {
            max-width: 100%;
            page-break-inside: avoid;
          }
          .footer {
            color: #666 !important;
          }
        }

        h1 {
          font-size: 36px;
          font-weight: 800;
          background: linear-gradient(135deg, #14B8A6, #3B82F6);
          -webkit-background-clip: text;
          background-clip: text;
          color: transparent;
          margin-bottom: 12px;
        }
        h2 {
          font-size: 28px;
          font-weight: 700;
          color: #14B8A6;
          margin: 0 0 8px 0;
          letter-spacing: -0.3px;
        }
        h3 {
          font-size: 20px;
          font-weight: 600;
          color: #F0F0F0;
          margin: 24px 0 12px 0;
          border-left: 4px solid #14B8A6;
          padding-left: 12px;
        }
        .header {
          display: flex;
          justify-content: space-between;
          align-items: center;
          margin-bottom: 40px;
          padding-bottom: 20px;
          border-bottom: 2px solid rgba(20,184,166,0.4);
        }
        .logo img {
          height: 52px;
          width: auto;
        }
        .date-badge {
          background: rgba(20,184,166,0.15);
          padding: 6px 14px;
          border-radius: 40px;
          font-size: 12px;
          font-weight: 500;
          color: #14B8A6;
          letter-spacing: 0.3px;
        }
        .summary-cards {
          display: flex;
          gap: 20px;
          margin: 32px 0 48px;
          flex-wrap: wrap;
        }
        .card {
          background: rgba(255,255,255,0.03);
          border: 1px solid rgba(255,255,255,0.08);
          border-radius: 28px;
          padding: 20px 24px;
          flex: 1;
          min-width: 180px;
          backdrop-filter: blur(2px);
          transition: transform 0.2s ease;
        }
        .card:hover {
          transform: translateY(-2px);
          background: rgba(255,255,255,0.05);
        }
        .card-title {
          font-size: 12px;
          text-transform: uppercase;
          letter-spacing: 1.2px;
          color: #9CA3AF;
          margin-bottom: 10px;
          font-weight: 500;
        }
        .card-value {
          font-size: 32px;
          font-weight: 700;
          color: #14B8A6;
          line-height: 1.2;
        }
        .positive { color: #10B981; }
        .negative { color: #F87171; }
        .product-list {
          margin: 40px 0 20px;
          display: flex;
          flex-wrap: wrap;
          gap: 12px;
        }
        .product-tag {
          background: rgba(20,184,166,0.12);
          border-radius: 40px;
          padding: 6px 18px;
          font-size: 13px;
          font-weight: 500;
          color: #14B8A6;
        }
        .product-page {
          page-break-before: always;
          margin-top: 20px;
        }
        table {
          width: 100%;
          border-collapse: collapse;
          margin: 20px 0 32px;
          font-size: 13px;
        }
        th {
          text-align: left;
          padding: 12px 10px;
          background: rgba(20,184,166,0.1);
          color: #14B8A6;
          font-weight: 600;
          border-bottom: 1px solid rgba(20,184,166,0.3);
        }
        td {
          padding: 10px;
          border-bottom: 1px solid rgba(255,255,255,0.06);
        }
        .footer {
          margin-top: 60px;
          text-align: center;
          font-size: 10px;
          color: #6B7280;
          border-top: 1px solid rgba(255,255,255,0.08);
          padding-top: 20px;
        }
        .chart-container {
          margin: 20px 0 30px;
          text-align: center;
          background: rgba(255,255,255,0.02);
          border-radius: 24px;
          padding: 16px;
        }
        canvas {
          max-width: 100%;
          height: auto;
        }
        .inline-stats {
          display: flex;
          gap: 24px;
          margin: 16px 0 8px;
          background: rgba(20,184,166,0.05);
          padding: 12px 20px;
          border-radius: 20px;
          flex-wrap: wrap;
        }
        .inline-stat {
          font-size: 13px;
        }
        .inline-stat strong {
          font-weight: 600;
          color: #14B8A6;
        }
        hr {
          border: none;
          height: 1px;
          background: rgba(255,255,255,0.1);
          margin: 20px 0;
        }
      </style>
    </head>
    <body>
      <!-- Cover / Front Page -->
      <div class="header">
        <div class="logo">
          <img src="https://res.cloudinary.com/dan4b75j7/image/upload/v1744200906/wealthblack/Welcome-Sankore_fy7xvu.png" alt="Sankore">
        </div>
        <div class="date-badge">As of ${reportDate}</div>
      </div>

      <h1>Investment Portfolio Report</h1>
      <p style="margin-bottom: 32px; color: #9CA3AF; font-size: 16px;">Comprehensive asset & performance analysis · All figures in USD</p>

      <div class="summary-cards">
        <div class="card">
          <div class="card-title">Total Beginning Value</div>
          <div class="card-value">${formatCurrency(totalBeginning)}</div>
        </div>
        <div class="card">
          <div class="card-title">Total Ending Value</div>
          <div class="card-value">${formatCurrency(totalEnding)}</div>
        </div>
        <div class="card">
          <div class="card-title">Net Gain / Loss</div>
          <div class="card-value ${totalGainLoss >= 0 ? 'positive' : 'negative'}">${formatSigned(totalGainLoss)}</div>
        </div>
        <div class="card">
          <div class="card-title">YTD Gain / Loss</div>
          <div class="card-value ${totalGainLossYTD >= 0 ? 'positive' : 'negative'}">${formatSigned(totalGainLossYTD)}</div>
        </div>
        <div class="card">
          <div class="card-title">Products Included</div>
          <div class="card-value">${assets.length}</div>
        </div>
      </div>

      <h2>Products in this Report</h2>
      <div class="product-list">
        ${assets.map(a => `<span class="product-tag">${a.product_type}</span>`).join('')}
      </div>
      <hr>

      <!-- Individual product pages -->
      ${assets.map((asset, idx) => {
        const productPerf = perfByProduct[asset.product_type] || [];
        const topHoldings = productPerf
          .map(h => ({ name: h.investment, value: h.end_value || 0 }))
          .sort((a, b) => b.value - a.value)
          .slice(0, 5);
        const holdingNames = topHoldings.map(h => h.name.length > 18 ? h.name.slice(0, 15) + '…' : h.name);
        const holdingValues = topHoldings.map(h => h.value);

        // Product-level summary
        const productEndTotal = productPerf.reduce((sum, p) => sum + (p.end_value || 0), 0);
        const productBegTotal = productPerf.reduce((sum, p) => sum + (p.beg_value || 0), 0);
        const productGain = productEndTotal - productBegTotal;

        return `
          <div class="product-page">
            <h2>${asset.product_type}</h2>
            <div class="inline-stats">
              <span class="inline-stat"><strong>Beginning Value</strong> ${formatCurrency(asset.beginning_value)}</span>
              <span class="inline-stat"><strong>Ending Value</strong> ${formatCurrency(asset.ending_value)}</span>
              <span class="inline-stat"><strong>Daily P&L</strong> <span class="${(asset.gain_loss || 0) >= 0 ? 'positive' : 'negative'}">${formatSigned(asset.gain_loss)}</span></span>
              <span class="inline-stat"><strong>YTD P&L</strong> <span class="${(asset.gain_loss_ytd || 0) >= 0 ? 'positive' : 'negative'}">${formatSigned(asset.gain_loss_ytd)}</span></span>
            </div>

            <div style="display: flex; flex-wrap: wrap; gap: 30px; margin: 20px 0;">
              <div style="flex: 1.2; min-width: 260px;">
                <h3>Holdings Breakdown</h3>
                <table style="margin-top: 0;">
                  <thead>
                    <tr><th>Investment</th><th>Beginning Value</th><th>Ending Value</th><th>Gain/Loss</th><th>Gain %</th></tr>
                  </thead>
                  <tbody>
                    ${productPerf.map(inv => `
                      <tr>
                        <td>${inv.investment}</td>
                        <td>${formatCurrency(inv.beg_value)}</td>
                        <td>${formatCurrency(inv.end_value)}</td>
                        <td class="${(inv.gain || 0) >= 0 ? 'positive' : 'negative'}">${formatSigned(inv.gain)}</td>
                        <td class="${(inv.gain_pct || 0) >= 0 ? 'positive' : 'negative'}">${inv.gain_pct ? inv.gain_pct + '%' : '—'}</td>
                      </tr>
                    `).join('')}
                  </tbody>
                </table>
              </div>
              <div class="chart-container" style="flex: 1; min-width: 280px;">
                <h3 style="margin-top: 0;">Top 5 Holdings by Value</h3>
                <canvas id="holdingChart_${idx}" width="500" height="280"></canvas>
              </div>
            </div>

            <div class="footer">
              ${asset.product_type} – Sankore Internal Platform · Generated on ${new Date().toLocaleString()}
            </div>

            <script>
              (function() {
                const canvas = document.getElementById('holdingChart_${idx}');
                if (canvas && typeof Chart !== 'undefined') {
                  new Chart(canvas.getContext('2d'), {
                    type: 'bar',
                    data: {
                      labels: ${JSON.stringify(holdingNames)},
                      datasets: [{
                        label: 'Ending Value (USD)',
                        data: ${JSON.stringify(holdingValues)},
                        backgroundColor: '${colors[idx % colors.length]}',
                        borderRadius: 8,
                        barPercentage: 0.7
                      }]
                    },
                    options: {
                      responsive: true,
                      maintainAspectRatio: true,
                      plugins: {
                        legend: { position: 'top', labels: { color: '#EAEAEF', font: { size: 11 } } },
                        tooltip: { callbacks: { label: (ctx) => \`\$\${ctx.raw.toLocaleString()}\` } }
                      },
                      scales: {
                        y: { beginAtZero: true, grid: { color: 'rgba(255,255,255,0.08)' }, ticks: { color: '#EAEAEF', callback: (val) => \$\${val.toLocaleString()} } },
                        x: { ticks: { color: '#EAEAEF', maxRotation: 35, minRotation: 25 } }
                      }
                    }
                  });
                }
              })();
            </script>
          </div>
        `;
      }).join('')}

      <div class="footer" style="margin-top: 40px;">
        Sankore Investment Report · ${reportDate} · All figures are unaudited estimates provided for informational purposes only.
      </div>
    </body>
    </html>
  `;
}
